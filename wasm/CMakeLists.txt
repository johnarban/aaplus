cmake_minimum_required(VERSION 3.10)
project(aaplus_wasm LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(AAPLUS_ROOT "${CMAKE_CURRENT_LIST_DIR}/..")
set(BINDINGS_DIR "${CMAKE_CURRENT_LIST_DIR}/bindings")

file(GLOB AAPLUS_SOURCES
  "${AAPLUS_ROOT}/AA*.cpp"
)

list(REMOVE_ITEM AAPLUS_SOURCES
  "${AAPLUS_ROOT}/AATest.cpp"
  "${AAPLUS_ROOT}/SunRiseSet.cpp"
)

add_executable(aaplus_wasm
  "${BINDINGS_DIR}/aa_bindings.cpp"
  ${AAPLUS_SOURCES}
)

target_include_directories(aaplus_wasm PRIVATE
  "${AAPLUS_ROOT}"
  "${BINDINGS_DIR}"
)

set_target_properties(aaplus_wasm PROPERTIES
  OUTPUT_NAME "aaplus"
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_LIST_DIR}/dist"
)

target_link_options(aaplus_wasm PRIVATE
  "SHELL:-s WASM=1"
  "SHELL:-s MODULARIZE=1"
  "SHELL:-s EXPORT_NAME='createAAPlusModule'"
  "SHELL:-s EXPORTED_RUNTIME_METHODS=['ccall','cwrap']"
  "SHELL:-s ALLOW_MEMORY_GROWTH=1"
  "SHELL:-s ENVIRONMENT=web,node"
  "SHELL:-s EXPORT_ES6=1"
  "SHELL:--bind"
  "SHELL:-O3"
)

# Build AATest as a standalone WASM executable for testing
# Generated by Copilot
add_executable(aatest_wasm
  "${AAPLUS_ROOT}/AATest.cpp"
  ${AAPLUS_SOURCES}
)

target_include_directories(aatest_wasm PRIVATE
  "${AAPLUS_ROOT}"
)

# Define empty stdafx.h include (AATest.cpp includes it but we don't need precompiled headers)
# Skip slow ELP2000 and ELPMPP02 algorithms for faster testing
target_compile_definitions(aatest_wasm PRIVATE
  AAPLUS_EXT_CLASS=
  AAPLUS_NO_ELP2000
  AAPLUS_NO_ELPMPP02
)

set_target_properties(aatest_wasm PROPERTIES
  OUTPUT_NAME "aatest"
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_LIST_DIR}/dist"
)

target_link_options(aatest_wasm PRIVATE
  "SHELL:-s WASM=1"
  "SHELL:-s ALLOW_MEMORY_GROWTH=1"
  "SHELL:-s ENVIRONMENT=node"
  "SHELL:-s EXIT_RUNTIME=1"
  "SHELL:-O2"
)
